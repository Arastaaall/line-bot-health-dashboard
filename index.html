// 8軸レーダーチャート（PFC + 食物繊維 + ビタミン類 + 筋肥大必須ミネラル） function renderRadarChart(stats, ideal) { const ctx = document.getElementById('radarChart').getContext('2d'); if (radarChartInstance) radarChartInstance.destroy(); // 各項目の達成率(%)計算 const pPct = Math.round((stats.avgProtein / ideal.protein) * 100) || 0; const fPct = Math.round((stats.avgFat / ideal.fat) * 100) || 0; const cPct = Math.round((stats.avgCarbs / ideal.carbs) * 100) || 0; const fiberPct = Math.round(((stats.avgFiber || 14) / ideal.fiber) * 100) || 0; // ビタミン類（統合スコア・達成率） const vitPct = stats.avgVitamins || 80; // 筋肥大特化ミネラル const zincPct = Math.round(((stats.avgZinc || 8) / (ideal.zinc || 10)) * 100) || 0; const magPct = Math.round(((stats.avgMagnesium || 260) / (ideal.magnesium || 320)) * 100) || 0; const sodiumPct = Math.round(((stats.avgSodium || 7) / (ideal.sodium || 8)) * 100) || 0; radarChartInstance = new Chart(ctx, { type: 'radar', data: { labels: [ 'たんぱく質', '脂質', '炭水化物', '食物繊維', 'ビタミン類', '亜鉛', 'マグネシウム', '塩分(ナトリウム)' ], datasets: [ { label: '過去7日間の平均摂取量', data: [pPct, fPct, cPct, fiberPct, vitPct, zincPct, magPct, sodiumPct], borderColor: '#3B82F6', backgroundColor: 'rgba(59, 130, 246, 0.2)', pointBackgroundColor: '#3B82F6', borderWidth: 2 }, { label: '理想値 (目標100%)', data: [100, 100, 100, 100, 100, 100, 100, 100], borderColor: '#9CA3AF', borderWidth: 1.5, borderDash: [4, 4], pointRadius: 0, fill: false } ] }, options: { responsive: true, maintainAspectRatio: false, scales: { r: { beginAtZero: true, max: 140, ticks: { stepSize: 20, display: false } } } } }); }
